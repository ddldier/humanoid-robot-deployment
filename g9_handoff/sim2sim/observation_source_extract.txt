   207	    def _adaptive_sampling(self, env_ids: Sequence[int]):
   208	        episode_failed = self._env.termination_manager.terminated[env_ids]
   209	        if torch.any(episode_failed):
   210	            current_bin_index = torch.clamp(
   211	                (self.time_steps * self.bin_count) // max(self.motion.time_step_total, 1), 0, self.bin_count - 1
   212	            )
   213	            fail_bins = current_bin_index[env_ids][episode_failed]
   214	            self._current_bin_failed[:] = torch.bincount(fail_bins, minlength=self.bin_count)
   215	
   216	        # Sample
   217	        sampling_probabilities = self.bin_failed_count + self.cfg.adaptive_uniform_ratio / float(self.bin_count)
   218	        sampling_probabilities = torch.nn.functional.pad(
   219	            sampling_probabilities.unsqueeze(0).unsqueeze(0),
   220	            (0, self.cfg.adaptive_kernel_size - 1),  # Non-causal kernel
   221	            mode="replicate",
   222	        )
   223	        sampling_probabilities = torch.nn.functional.conv1d(sampling_probabilities, self.kernel.view(1, 1, -1)).view(-1)
   224	
   225	        sampling_probabilities = sampling_probabilities / sampling_probabilities.sum()
   226	
   227	        sampled_bins = torch.multinomial(sampling_probabilities, len(env_ids), replacement=True)
   228	
   229	        self.time_steps[env_ids] = (
   230	            (sampled_bins + sample_uniform(0.0, 1.0, (len(env_ids),), device=self.device))
   231	            / self.bin_count
   232	            * (self.motion.time_step_total - 1)
   233	        ).long()
   234	
   235	        # Metrics
   236	        H = -(sampling_probabilities * (sampling_probabilities + 1e-12).log()).sum()
   237	        H_norm = H / math.log(self.bin_count)
   238	        pmax, imax = sampling_probabilities.max(dim=0)
   239	        self.metrics["sampling_entropy"][:] = H_norm
   240	        self.metrics["sampling_top1_prob"][:] = pmax
